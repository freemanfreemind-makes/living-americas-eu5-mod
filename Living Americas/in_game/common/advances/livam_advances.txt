war_bands_livam = {
	age = age_1_traditions
	icon = warriors_unity

	potential = {
		government_type = government_type:tribe
		country_type = army
		exists = capital
		NOT = { is_subject_type = colonial_nation }
		original_capital ?= {
			OR = {
			sub_continent = sub_continent:north_america
			sub_continent = sub_continent:south_america
			}
		}
	}

	requires = metallurgy

	army_light_infantry_reinforce_cost_modifier = -0.33
	army_light_infantry_maintenance_cost_modifier = -0.33

	army_heavy_infantry_reinforce_cost_modifier = -0.33
	army_heavy_infantry_maintenance_cost_modifier = -0.33

	always_allow_army_levies = yes

	#starting_technology_level = 1
}

warrior_society_advance_livam = {
	age = age_1_traditions
	icon = warrior_lodge

	potential = {
		government_type = government_type:tribe
		country_type = army
		exists = capital
		NOT = { is_subject_type = colonial_nation }
		original_capital ?= {
			OR = {
			sub_continent = sub_continent:north_america
			sub_continent = sub_continent:south_america
			}
		}
	}

	requires = war_bands_livam

	unlock_building = warrior_society
	levy_maintenance_modifier = -0.5
	enable_taxation = yes
	tribes_estate_min_tax = 0.15
	tribes_estate_max_tax = 0.20

	#starting_technology_level = 1
}
